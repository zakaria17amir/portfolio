# Portfolio Site Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and publish a static, dependency-free portfolio site at `https://zakaria17amir.github.io/portfolio` that serves a non-technical HR screener in 20 seconds and a skeptical technical lead in 5 minutes.

**Architecture:** Hand-written semantic HTML, styled by Tailwind CSS v4 compiled ahead of time into a committed stylesheet. No runtime framework, no runtime dependencies, no build step on the server — GitHub Pages serves the files as-is. JavaScript is progressive enhancement only (theme toggle, mobile nav, email assembly, scroll reveal); every piece of content and navigation works with JS disabled.

**Tech Stack:** HTML5, Tailwind CSS v4 (`@tailwindcss/cli`), vanilla ES modules, GitHub Pages, `gh` CLI.

**Spec:** `docs/superpowers/specs/2026-09-16-portfolio-design.md`

## Global Constraints

- **Repo:** `zakaria17amir/portfolio`, public, branch `main`, Pages from `/` (root).
- **Public email:** `mailtoamir17@gmail.com`. The university address `amir@student.elte.hu` must NOT appear anywhere on the site.
- **No phone number anywhere in the site source.** `+36 202 608 876` must never appear in any file in this repo.
- **Commit authorship:** every commit authored and committed as `zakaria17amir <acnwre@gmail.com>`. No `Co-Authored-By` trailer. No bot co-author. No tool attribution in any commit message.
- **No runtime dependencies.** `node_modules/` is gitignored; `assets/css/style.css` is built and committed.
- **No third-party scripts, fonts, analytics, or cookies.**
- **Every project card links to a real public repo** under `github.com/zakaria17amir`.
- **Accessibility:** WCAG AA contrast in both themes, one `<h1>` per page, visible focus rings, skip link, semantic landmarks.
- **Performance budget:** under 150 KB transferred per page excluding the CV PDF; Lighthouse >= 95 in all four categories.
- **All internal links relative** so a custom domain can be added later without edits.
- Content facts (dates, GPA, employers, bullet wording) come from `CV_Amir_MSCI_DataScience.pdf` and `CV_Amir_TCS_GenerativeAI.pdf` and must not be embellished beyond what those state.

```js
/**
 * Progressive enhancement only. Every feature here is optional: with this file
 * blocked or failing, the site still reads, navigates and links correctly.
 */

/** Theme toggle. The initial class is set by the inline script in <head>. */
function initTheme() {
  const toggle = document.querySelector("#theme-toggle");
  if (!toggle) return;

  const sync = () => {
    const isDark = document.documentElement.classList.contains("dark");
    toggle.setAttribute("aria-pressed", String(isDark));
  };

  sync();

  toggle.addEventListener("click", () => {
    const isDark = document.documentElement.classList.toggle("dark");
    try {
      localStorage.setItem("theme", isDark ? "dark" : "light");
    } catch (e) {
      /* storage blocked: the toggle still works for this page view */
    }
    sync();
  });
}

/** Mobile navigation. */
function initNav() {
  const toggle = document.querySelector("#nav-toggle");
  const nav = document.querySelector("#site-nav");
  if (!toggle || !nav) return;

  const setOpen = (open) => {
    nav.classList.toggle("hidden", !open);
    toggle.setAttribute("aria-expanded", String(open));
    toggle.setAttribute("aria-label", open ? "Close menu" : "Open menu");
  };

  toggle.addEventListener("click", () => {
    setOpen(toggle.getAttribute("aria-expanded") !== "true");
  });

  // Close after following an in-page link, and on Escape.
  nav.addEventListener("click", (e) => {
    if (e.target.closest("a")) setOpen(false);
  });
  document.addEventListener("keydown", (e) => {
    if (e.key === "Escape" && toggle.getAttribute("aria-expanded") === "true") {
      setOpen(false);
      toggle.focus();
    }
  });
}

/**
 * Assemble the email address at runtime so it is not sitting in the HTML as a
 * scrapeable plain string. Anchors carry a visible fallback until this runs.
 */
function initEmail() {
  // Duplicated from `profile.emailParts` in content.js on purpose: importing
  // content.js here would cost an extra module fetch for two short strings.
  // If you change the address, change it in BOTH files.
  const user = "mailtoamir17";
  const domain = "gmail.com";
  const address = `${user}@${domain}`;

  document.querySelectorAll("[data-email]").forEach((el) => {
    el.setAttribute("href", `mailto:${address}`);
    if (el.dataset.email === "text") el.textContent = address;
  });
}

/** Fade sections in as they enter the viewport. */
function initReveal() {
  const targets = document.querySelectorAll("[data-reveal]");
  if (!targets.length) return;

  const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
  if (reduced || !("IntersectionObserver" in window)) return;

  // Hide only now that we know the observer will run to un-hide them.
  targets.forEach((el) => el.classList.add("reveal"));

  const observer = new IntersectionObserver(
    (entries) => {
      entries.forEach((entry) => {
        if (!entry.isIntersecting) return;
        entry.target.classList.add("is-visible");
        observer.unobserve(entry.target);
      });
    },
    { rootMargin: "0px 0px -10% 0px", threshold: 0.05 }
  );

  targets.forEach((el) => observer.observe(el));
}

/** Highlight the nav link for the section currently on screen. */
function initActiveNav() {
  const links = document.querySelectorAll('#site-nav a[href^="#"]');
  if (!links.length || !("IntersectionObserver" in window)) return;

  const byId = new Map();
  links.forEach((link) => byId.set(link.getAttribute("href").slice(1), link));

  const observer = new IntersectionObserver(
    (entries) => {
      entries.forEach((entry) => {
        const link = byId.get(entry.target.id);
        if (link) link.classList.toggle("text-accent", entry.isIntersecting);
      });
    },
    { rootMargin: "-40% 0px -55% 0px" }
  );

  byId.forEach((_, id) => {
    const section = document.getElementById(id);
    if (section) observer.observe(section);
  });
}

initTheme();
initNav();
initEmail();
initReveal();
initActiveNav();
```
